cmake_minimum_required(VERSION 3.24)
project(DataCopy VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "DataCopy requires Windows 10 or later")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(datacopy_core src/core.cpp src/platform.cpp)
target_include_directories(datacopy_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_definitions(datacopy_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(datacopy_core PUBLIC shell32 ole32 setupapi cfgmgr32)
add_executable(DataCopy WIN32 src/main.cpp)
target_link_libraries(DataCopy PRIVATE datacopy_core comctl32 comdlg32 gdi32 user32)
if(MSVC)
  target_compile_options(datacopy_core PRIVATE /W4 /utf-8 /permissive-)
  target_compile_options(DataCopy PRIVATE /W4 /utf-8 /permissive-)
  target_link_options(DataCopy PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest")
endif()
enable_testing()
add_executable(datacopy_tests tests/core_tests.cpp)
target_link_libraries(datacopy_tests PRIVATE datacopy_core)
if(MSVC)
  target_compile_options(datacopy_tests PRIVATE /W4 /utf-8 /permissive-)
endif()
add_test(NAME transfer_safety COMMAND datacopy_tests)
set_tests_properties(transfer_safety PROPERTIES TIMEOUT 180)
